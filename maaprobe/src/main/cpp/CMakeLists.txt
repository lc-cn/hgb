cmake_minimum_required(VERSION 3.22)
project(YysMaaProbe LANGUAGES CXX)
add_library(MaaFramework SHARED IMPORTED)
set_target_properties(MaaFramework PROPERTIES IMPORTED_LOCATION "${MAA_ROOT}/jniLibs/${ANDROID_ABI}/libMaaFramework.so")
add_library(yys_maa_probe SHARED MaaProbe.cpp)
target_compile_features(yys_maa_probe PRIVATE cxx_std_17)
target_include_directories(yys_maa_probe PRIVATE "${MAA_ROOT}/include")
target_link_libraries(yys_maa_probe PRIVATE MaaFramework)
set_target_properties(yys_maa_probe PROPERTIES CXX_VISIBILITY_PRESET hidden)
