cmake_minimum_required(VERSION 3.22)
project(YysMaaRegionProbe LANGUAGES CXX)
add_library(MaaFramework SHARED IMPORTED)
set_target_properties(MaaFramework PROPERTIES IMPORTED_LOCATION "${MAA_ROOT}/jniLibs/${ANDROID_ABI}/libMaaFramework.so")
add_library(yys_maa_region_probe SHARED MaaRegionProbe.cpp)
target_compile_features(yys_maa_region_probe PRIVATE cxx_std_17)
target_include_directories(yys_maa_region_probe PRIVATE "${MAA_ROOT}/include")
target_link_libraries(yys_maa_region_probe PRIVATE MaaFramework)
set_target_properties(yys_maa_region_probe PROPERTIES CXX_VISIBILITY_PRESET hidden)
